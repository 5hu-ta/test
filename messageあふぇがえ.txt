local Players = game:GetService("Players")

local LocalPlayer = Players.LocalPlayer
local Character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
local HumanoidRootPart = Character:WaitForChild("HumanoidRootPart")
local Humanoid = Character:WaitForChild("Humanoid")

local function Notify(title, text, duration)
    StarterGui:SetCore("SendNotification", {
        Title = title,
        Text = text,
        Duration = duration or 3
    })
end

getgenv().AutoBounty = {
    Enabled = false,
    attackDelay = 0.11,
    skillDelay = 0.32,
    trackOffset = Vector3.new(0, 1.5, 4)
}

local currentTarget = nil
local lastAttack = 0
local lastSkill = 0
local skillKeys = {"Z", "X", "C", "V", "F"}

LocalPlayer.Idled:Connect(function()
    VirtualUser:CaptureController()
    VirtualUser:ClickButton2(Vector2.new())
end)

local function getTools()
    local tools = {melee = nil, fruit = nil, gun = nil}
    local sources = {LocalPlayer.Backpack, Character}
    for _, source in pairs(sources) do
        for _, tool in pairs(source:GetChildren()) do
            if tool:IsA("Tool") then
                local name = tool.Name:lower()
                if name:find("combat") or name:find("dark step") or name:find("electric") or name:find("water") or name:find("dragon") or name:find("superhuman") or name:find("death") or name:find("sharkman") or name:find("godhuman") or name:find("sanguine") or name:find("advanced") then
                    tools.melee = tool
                elseif tool:FindFirstChild("Fruit") or name:find("fruit") or name:find("dough") or name:find("kitsune") or name:find("dragon") or name:find("leopard") or name:find("control") or name:find("ice") or name:find("magma") or name:find("light") or name:find("dark") or name:find("buddha") or name:find("portal") or name:find("spirit") or name:find("shadow") or name:find("venom") or name:find("quake") or name:find("string") or name:find("rumble") or name:find("paw") or name:find("gravity") or name:find("phoenix") or name:find("magnet") or name:find("tiger") or name:find("gas") then
                    tools.fruit = tool
                elseif name:find("gun") or name:find("rifle") or name:find("pistol") or name:find("musket") or name:find("flintlock") or name:find("slingshot") or name:find("bazooka") or name:find("cannon") or name:find("kabucha") or name:find("skull") or name:find("venom bow") or name:find("acidum") or name:find("bizarre") or name:find("dragonstorm") then
                    tools.gun = tool
                end
            end
        end
    end
    return tools
end

local function equipTool(tool)
    if tool and tool.Parent ~= Character then
        pcall(function()
            Humanoid:EquipTool(tool)
        end)
    end
end

local function doM1()
    VirtualUser:CaptureController()
    VirtualUser:Button1Down(Vector2.new(0, 0))
    task.wait(0.04)
    VirtualUser:Button1Up(Vector2.new(0, 0))
end

local function doSkill(key)
    VirtualInputManager:SendKeyEvent(true, key, false, game)
    task.wait(0.04)
    VirtualInputManager:SendKeyEvent(false, key, false, game)
end

local function getNearestTarget()
    local nearest = nil
    local shortest = math.huge
    local myPos = HumanoidRootPart and HumanoidRootPart.Position or Vector3.zero

    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local hrp = player.Character:FindFirstChild("HumanoidRootPart")
            local hum = player.Character:FindFirstChild("Humanoid")
            if hrp and hum and hum.Health > 0 then
                local dist = (myPos - hrp.Position).Magnitude
                if dist < shortest then
                    shortest = dist
                    nearest = player
                end
            end
        end
    end
    return nearest
end

local connection
local function startLoop()
    if connection then connection:Disconnect() end
    connection = RunService.Heartbeat:Connect(function()
        if not getgenv().AutoBounty.Enabled then return end
        pcall(function()
            Character = LocalPlayer.Character
            if not Character then return end
            HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")
            Humanoid = Character:FindFirstChild("Humanoid")
            if not HumanoidRootPart or not Humanoid or Humanoid.Health <= 0 then return end

            currentTarget = getNearestTarget()
            if currentTarget and currentTarget.Character then
                local targetHRP = currentTarget.Character:FindFirstChild("HumanoidRootPart")
                local targetHum = currentTarget.Character:FindFirstChild("Humanoid")
                if targetHRP and targetHum and targetHum.Health > 0 then
                    -- Unlimited track
                    local lookAt = CFrame.new(HumanoidRootPart.Position, targetHRP.Position)
                    HumanoidRootPart.CFrame = lookAt * CFrame.new(0, 0, -getgenv().AutoBounty.trackOffset.Z) + Vector3.new(0, getgenv().AutoBounty.trackOffset.Y, 0)

                    local now = tick()
                    local tools = getTools()

                    if now - lastAttack >= getgenv().AutoBounty.attackDelay then
                        if tools.melee then equipTool(tools.melee) end
                        doM1()
                        if tools.fruit then
                            equipTool(tools.fruit)
                            doM1()
                        end
                        if tools.gun then
                            equipTool(tools.gun)
                            doM1()
                        end
                        lastAttack = now
                    end

                    if now - lastSkill >= getgenv().AutoBounty.skillDelay then
                        for _, key in ipairs(skillKeys) do
                            doSkill(key)
                            task.wait(0.06)
                        end
                        lastSkill = now
                    end
                end
            else
                currentTarget = nil
            end
        end)
    end)
end
